Interface RFID with ESP32

RC522 RFID Wiring

RC522 Pin ESP32 Pin
SDA GPIO5
SCK GPIO18
MOSI GPIO23
MISO GPIO19
RST GPIO4
VCC 3.3V
GND GND

LED Connection:

  • Anode (long leg) → 220Ω resistor → GPIO2
  • Cathode → GND

Buzzer Connection:

  • Positive (+) → GPIO15
  • Negative (–) → GND

Basic RFID Code

#include <SPI.h>  
#include <MFRC522.h>

#define SS_PIN 5  
#define RST_PIN 4  
#define LED_PIN 2  
#define BUZZER_PIN 15

MFRC522 mfrc522(SS_PIN, RST_PIN);

byte validUID[4] = {0xDE, 0xAD, 0xBE, 0xEF};

void setup() {  
  Serial.begin(115200);  
  SPI.begin(); 
  mfrc522.PCD_Init();  
  pinMode(LED_PIN, OUTPUT);  
  pinMode(BUZZER_PIN, OUTPUT); 
}

void loop() {  
  if (!mfrc522.PICC_IsNewCardPresent() || !mfrc522.PICC_ReadCardSerial()) { 
    return; 
  }

  if (isUIDMatch(mfrc522.uid.uidByte, validUID)) { 
    blinkAndBeep(LED_PIN, BUZZER_PIN, 1, 200);  
    Serial.println("Valid User"); 
  } else { 
    blinkAndBeep(LED_PIN, BUZZER_PIN, 3, 200);  
    Serial.println("Unknown User"); 
  }

  mfrc522.PICC_HaltA(); 
}

bool isUIDMatch(byte *a, byte *b) {  
  for (byte i = 0; i < 4; i++) {  
    if (a[i] != b[i]) return false; 
  } 
  return true; 
}

void blinkAndBeep(int ledPin, int buzzerPin, int times, int delayTime) {
  for (int i = 0; i < times; i++) {
    digitalWrite(ledPin, HIGH);  
    digitalWrite(buzzerPin, HIGH);  
    delay(delayTime); 
    digitalWrite(ledPin, LOW);  
    digitalWrite(buzzerPin, LOW);  
    delay(delayTime); 
  } 
}

RTC Module Integration

DS1302 RTC Connection

DS1302 RTC Module Connections:

  • VCC → 3.3V
  • GND → GND
  • CLK → GPIO14
  • DAT → GPIO27
  • RST → GPIO26

RTC Code Implementation

#include <DS1302.h>

// RTC pins
const int kCePin   = 26;  // Chip Enable
const int kIoPin   = 27;  // Input/Output
const int kSclkPin = 14;  // Serial Clock

DS1302 rtc(kCePin, kIoPin, kSclkPin);

void setup() {
  Serial.begin(115200);
  
  // Set the clock to run-mode, and disable the write protection
  rtc.halt(false);
  rtc.writeProtect(false);
  
  // Set initial time (only needed once)
  // rtc.setDateTime(2024, 1, 15, 10, 30, 0);
}

String getCurrentTime() {
  Time t = rtc.getTime();
  
  String timeStr = "";
  timeStr += String(t.date) + "/";
  timeStr += String(t.mon) + "/";
  timeStr += String(t.year) + " ";
  timeStr += String(t.hour) + ":";
  timeStr += String(t.min) + ":";
  timeStr += String(t.sec);
  
  return timeStr;
}

ESP32 Code for Google Sheets

#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>

const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";
const char* scriptURL = "YOUR_GOOGLE_SCRIPT_URL";

void sendToGoogleSheets(String uid, String status, String timestamp) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(scriptURL);
    http.addHeader("Content-Type", "application/json");
    
    DynamicJsonDocument doc(1024);
    doc["uid"] = uid;
    doc["status"] = status;
    doc["timestamp"] = timestamp;
    
    String jsonString;
    serializeJson(doc, jsonString);
    
    int httpResponseCode = http.POST(jsonString);
    
    if (httpResponseCode > 0) {
      String response = http.getString();
      Serial.println("Data sent successfully");
    } else {
      Serial.println("Error sending data");
    }
    
    http.end();
  }
}

System Flowchart

System Operation Flow

  1. System initialization and Wi-Fi connection
  2. Display "Scan your card" on LCD
  3. Wait for RFID card detection
  4. Read card UID when detected
  5. Compare UID with authorized cards
  6. If authorized: Grant access, open gate, log to Google Sheets
  7. If unauthorized: Deny access, sound alarm, log attempt
  8. Return to scanning mode

Complete System Code

#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <SPI.h>
#include <MFRC522.h>
#include <ESP32Servo.h>
#include <DS1302.h>

// Pin Definitions
#define SS_PIN 5
#define RST_PIN 4
#define LED_PIN 2
#define BUZZER_PIN 15
#define SERVO_PIN 13

// RTC pins
const int kCePin = 26;
const int kIoPin = 27;
const int kSclkPin = 14;

// Initialize components
LiquidCrystal_I2C lcd(0x27, 16, 2);
MFRC522 mfrc522(SS_PIN, RST_PIN);
Servo gateServo;
DS1302 rtc(kCePin, kIoPin, kSclkPin);

// Network credentials
const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";
const char* scriptURL = "YOUR_GOOGLE_SCRIPT_URL";

// Authorized UIDs
byte authorizedUIDs[][4] = {
  {0xED, 0x0D, 0x0C, 0x32},
  {0x12, 0x34, 0x56, 0x78}
};
int numAuthorizedCards = 2;

void setup() {
  Serial.begin(115200);
  
  // Initialize components
  SPI.begin();
  mfrc522.PCD_Init();
  
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  
  gateServo.attach(SERVO_PIN);
  gateServo.write(0); // Gate closed
  
  Wire.begin(21, 22);
  lcd.begin();
  lcd.backlight();
  
  rtc.halt(false);
  rtc.writeProtect(false);
  
  // Connect to WiFi
  WiFi.begin(ssid, password);
  lcd.setCursor(0, 0);
  lcd.print("Connecting WiFi");
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.print(".");
  }
  
  Serial.println("WiFi connected!");
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("System Ready");
  delay(2000);
  
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Scan your card");
}

void loop() {
  if (!mfrc522.PICC_IsNewCardPresent() || !mfrc522.PICC_ReadCardSerial()) {
    return;
  }
  
  String uidString = "";
  bool isAuthorized = false;
  
  // Read UID and check authorization
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    uidString += String(mfrc522.uid.uidByte[i], HEX);
    if (i < mfrc522.uid.size - 1) uidString += ":";
  }
  
  // Check if card is authorized
  for (int i = 0; i < numAuthorizedCards; i++) {
    bool match = true;
    for (byte j = 0; j < 4; j++) {
      if (mfrc522.uid.uidByte[j] != authorizedUIDs[i][j]) {
        match = false;
        break;
      }
    }
    if (match) {
      isAuthorized = true;
      break;
    }
  }
  
  String timestamp = getCurrentTime();
  String status = isAuthorized ? "Granted" : "Denied";
  
  // Display result
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Access: " + status);
  
  if (isAuthorized) {
    digitalWrite(LED_PIN, HIGH);
    digitalWrite(BUZZER_PIN, HIGH);
    delay(500);
    digitalWrite(LED_PIN, LOW);
    digitalWrite(BUZZER_PIN, LOW);
    
    // Open gate
    gateServo.write(90);
    delay(3000);
    gateServo.write(0);
    
  } else {
    for (int i = 0; i < 3; i++) {
      digitalWrite(LED_PIN, HIGH);
      digitalWrite(BUZZER_PIN, HIGH);
      delay(200);
      digitalWrite(LED_PIN, LOW);
      digitalWrite(BUZZER_PIN, LOW);
      delay(200);
    }
  }
  
  // Send to Google Sheets
  sendToGoogleSheets(uidString, status, timestamp);
  
  mfrc522.PICC_HaltA();
  mfrc522.PCD_StopCrypto1();
  
  delay(2000);
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Scan your card");
}

String getCurrentTime() {
  Time t = rtc.getTime();
  String timeStr = "";
  timeStr += String(t.date) + "/" + String(t.mon) + "/" + String(t.year);
  timeStr += " " + String(t.hour) + ":" + String(t.min) + ":" + String(t.sec);
  return timeStr;
}

void sendToGoogleSheets(String uid, String status, String timestamp) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(scriptURL);
    http.addHeader("Content-Type", "application/json");
    
    DynamicJsonDocument doc(1024);
    doc["uid"] = uid;
    doc["status"] = status;
    doc["timestamp"] = timestamp;
    
    String jsonString;
    serializeJson(doc, jsonString);
    
    int httpResponseCode = http.POST(jsonString);
    
    if (httpResponseCode > 0) {
      Serial.println("Data logged successfully");
    } else {
      Serial.println("Error logging data");
    }
    
    http.end();
  }
}
